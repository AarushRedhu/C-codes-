Enter 5 values of array from user, enter a value from user and search this value in array, if value is found then print “found” or not then print “not found”

CODE

#include <stdio.h>
void main()
{
int a[5], i, n, found = 0;
printf("Enter 5 numbers:\n");
for (i = 0; i < 5; i++)
{
scanf("%d", &a[i]);
}
printf("Enter number to search: ");
scanf("%d", &n);
for (i = 0; i < 5; i++)
{
if (a[i] == n)
{
found = 1;
}
}
if (found == 1)
printf("Found");
else
printf("Not found");
}

OUTPUT

Enter 5 numbers:
10
20
30
40
50
Enter number to search: 60
Not found

Enter five values of array and sort in ascending order

CODE

#include <stdio.h> void main()
{
int a[5], i, j, temp;
printf("Enter 5 numbers:\n");
for (i = 0; i < 5; i++)
{
scanf("%d", &a[i]);
}
for (i = 0; i < 5; i++)
{
for (j = i + 1; j < 5; j++)
{
if (a[i] > a[j])
{
temp = a[i];
a[i] = a[j];
a[j] = temp;
}
}
}
printf("Ascending order:\n");
for (i = 0; i < 5; i++)
{
printf("%d\n", a[i]);
}
}

OUTPUT

Enter 5 numbers:
50
20
40
10
30
Ascending order:
10
20
30
40
50

Find maximum values of array

CODE

#include <stdio.h>
void main()
{
int a[5], i, max;
printf("Enter 5 numbers:\n");
for (i = 0; i < 5; i++)
{
scanf("%d", &a[i]);
}
max = a[0];
for (i = 1; i < 5; i++)
{
if (a[i] > max)
{
max = a[i];
}
}
printf("Maximum value = %d", max);
}

OUTPUT

Enter 5 numbers:
10
50
30
20
40
Maximum value = 50
