Enter marks of students and calculate average of student’s marks ?

CODE

#include <stdio.h>
int main()
{
int m1, m2, m3, m4, m5, average;
printf("Enter marks of five subjects (out of 100): ");
scanf("%d %d %d %d %d", &m1, &m2, &m3, &m4, &m5);
if (m1 >= 0 && m1 <= 100 &&
m2 >= 0 && m2 <= 100 &&
m3 >= 0 && m3 <= 100 &&
m4 >= 0 && m4 <= 100 &&
m5 >= 0 && m5 <= 100)
{
average = (m1 + m2 + m3 + m4 + m5) / 5;
printf("Average marks = %d", average);
}
else
{ printf("Invalid marks! Enter marks between 0 and 100.");
}
return 0;
}

OUTPUT

Enter marks of five subjects (out of 100): 85 90 78 92 88
Average marks = 86

Calculate simple interest = (P x R x T)/ 100

CODE

#include <stdio.h>
int main()
{
int p, r, t, si;
printf("Enter principal amount: ");
scanf("%d", &p);
printf("Enter rate of interest: ");
scanf("%d", &r);
printf("Enter time in years: ");
scanf("%d", &t);
si = (p * r * t) / 100;
printf("Simple Interest = %d", si);
return 0;
}

OUTPUT

Enter principal amount: 5000
Enter rate of interest: 5
Enter time in years: 2
Simple Interest = 500
