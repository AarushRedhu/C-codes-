Write a program of recursion with global variable and display the output

CODE

#include <stdio.h>
#include <conio.h>
int num;
int fact()
{
if(num == 1)
{
return 1;
}
else
{
return num * fact(--num);
}
}
void main()
{
int ans;
clrscr();
printf("Enter a number: ");
scanf("%d", &num);
ans = fact();
printf("Factorial = %d", ans);
getch();
}

OUTPUT

Enter a number: 4
Factorial = 24

Write a program of recursion with local variable and display the output

CODE

#include <stdio.h>
#include <conio.h>
int sum(int n)
{
int result;
if(n == 1)
{
return 1;
}
else
{
result = n + sum(n - 1);
return result;
}
}
void main()
{
int num, ans;
clrscr();
printf("Enter a number: ");
scanf("%d", &num);
ans = sum(num);
printf("Sum = %d", ans);
getch();
}

OUTPUT

Enter a number: 4
Sum = 10
