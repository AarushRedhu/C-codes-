POINTERS

Write a program in which a pointer points to an array and insert value from user through pointer

CODE

#include <stdio.h>
void main()
{
int a[5], i;
int *p;
p = a;
for(i = 1; i <= 5; i++)
{
printf("Enter %d value: ", i);
scanf("%d", p);
p++;
}
p = a;
printf("\nArray values: ");
for(i = 1; i <= 5; i++)
{
printf("%d ", *p);
p++;
}
}

OUTPUT

Enter 1 value: 55
Enter 2 value: 66
Enter 3 value: 77
Enter 4 value: 88
Enter 5 value: 99
Array values: 55 66 77 88 99

Write a program to input and display the elements of a 2D array of size 2 x 3 using a pointer

CODE

#include <stdio.h>
void main()
{
int a[2][3], i, j;
int *p;
p = a;
for(i = 1; i <= 6; i++)
{
printf("Enter %d value: ", i);
scanf("%d", p);
p++;
}
p = a;
printf("\nArray values: \n");
for(i = 0; i < 2; i++)
{
for(j = 0; j < 3; j++)
{
printf("%d ",*p);
p++;
}
printf("\n");
}
}

OUTPUT

Enter 1 value: 22
Enter 2 value: 33
Enter 3 value: 44
Enter 4 value: 55
Enter 5 value: 66
Enter 6 value: 77
Array values:
22 33 44
55 66 77
