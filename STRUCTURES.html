Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form

CODE

#include <stdio.h>
#include <conio.h>
struct employee
{
int id;
char name[30];
char designation[30];
};
void main()
{
struct employee work[50];
int i;
clrscr();
for(i = 0; i < 50; i++)
{
printf("\nEnter details of employee %d\n", i + 1);
printf("Enter ID: ");
scanf("%d", &work[i].id);
printf("Enter Name: ");
scanf(" %[^\n]", work[i].name);
printf("Enter Designation: ");
scanf(" %[^\n]", work[i].designation);
}
printf("\n\nEMPLOYEE DETAILS\n");
printf("-----------------------------------------------\n");
printf("ID\tName\t\tDesignation\n");
printf("-----------------------------------------------\n");
for(i = 0; i < 50; i++)
{
printf("%d\t%-15s\t%s\n",
work[i].id, work[i].name, work[i].designation);
}
}

OUTPUT

EMPLOYEE DETAILS
-----------------------------------------------
ID Name Designation
-----------------------------------------------
101 Rahul Manager
102 Aman Accountant
103 Priya Developer

Enter student details like roll name, fee, DOB and print all the detail by ascending order of roll number

CODE

#include <stdio.h>
#include <conio.h>
struct student
{
int roll;
char name[30];
float fee;
char dob[15];
};
void main()
{
struct student s[5], temp;
int i, j;
clrscr();
for(i = 0; i < 5; i++)
{
printf("\nEnter details of student %d\n", i + 1);
printf("Enter Roll Number: ");
scanf("%d", &s[i].roll);
printf("Enter Name: ");
scanf(" %[^\n]", s[i].name);
printf("Enter Fee: ");
scanf("%f", &s[i].fee);
printf("Enter DOB (DD/MM/YYYY): ");
scanf("%s", s[i].dob);
}
for(i = 0; i < 5; i++)
{
for(j = i + 1; j < 5; j++)
{
if(s[i].roll > s[j].roll)
{
temp = s[i];
s[i] = s[j];
s[j] = temp;
}
}
}
printf("\nSTUDENT DETAILS IN ASCENDING ORDER\n");
printf("-----------------------------------------------\n");
printf("Roll\tName\t\tFee\tDOB\n");
printf("-----------------------------------------------\n");
for(i = 0; i < 5; i++)
{ printf("%d\t%-15s\t%.2f\t%s\n", s[i].roll, s[i].name, s[i].fee, s[i].dob);
} }
