Enter 2 matrix from user and multiply them

CODE

#include <stdio.h>
#include <conio.h>
void main()
{
int a[2][2], b[2][2], x[2][2];
int i, j, k;
clrscr();
printf("Enter elements of first matrix:\n");
for(i = 0; i < 2; i++)
{
for(j = 0; j < 2; j++)
{
scanf("%d", &a[i][j]);
}
}
printf("Enter elements of second matrix:\n");
for(i = 0; i < 2; i++)
{
for(j = 0; j < 2; j++)
{
scanf("%d", &b[i][j]);
}
}
for(i = 0; i < 2; i++)
{
for(j = 0; j < 2; j++)
{
x[i][j] = 0;
for(k = 0; k < 2; k++)
{
x[i][j] = x[i][j] + a[i][k] * b[k][j];
}
}
}
printf("\nMultiplication of two matrices:\n");
for(i = 0; i < 2; i++)
{
for(j = 0; j < 2; j++)
{
printf("%d\t", x[i][j]);
}
printf("\n");
}
}

3 4
Enter elements of second matrix:
