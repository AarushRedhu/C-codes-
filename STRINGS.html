CODE

#include <conio.h>

{

int i;
clrscr();

gets(str);

{
if(i == 0 && str[i] >= 'a' && str[i] <= 'z')

}

String palindrome example (using reverse)

CODE

#include<stdio.h>
#include<conio.h>
#include<string.h>
void main()
{
char str[20], rev[20];
int i,j;
printf("Enter a string: ");
gets(str);
j=0;
for
(i=strlen(str)-1;i>=0;i--)
{
rev[j]=str[i];
j++;
}
rev[j]='\0';
if(strcmp(str,rev)==0)
printf("Palindrome");
else
printf("Not Palindrome");
}
