CODE

#include <stdio.h>
#include <conio.h>
#include <string.h>
void main()
{
FILE *fp;
char fname[50], data[100];
int i;
clrscr();
printf("Enter file name: ");
scanf("%s", fname);
fp = fopen(fname, "w");
printf("Enter data: ");
scanf(" %[^\n]", data);
for(i = 0; data[i] != '\0'; i++)
{
if(data[i] >= 97 && data[i] <= 122)
{
data[i] = data[i] - 32;
}
}
fprintf(fp, "%s", data);
fclose(fp);
printf("Data stored in capital letters.");
}

OUTPUT

Enter file name: code
Enter data: Iam bca student
Data stored in capital letters.

Enter the file name from the user and the data from the user till the user wants to enter.

CODE

#include <stdio.h>
#include <conio.h>
void main()
{
FILE *fp;
char fname[50], data[100];
char choice;
clrscr();
printf("Enter file name: ");
scanf("%s", fname);
fp = fopen(fname, "w");
do
{
printf("Enter data: ");
scanf(" %[^\n]", data);
fprintf(fp, "%s\n", data);
printf("Do you want to enter more data? (y/n): ");
scanf(" %c", &choice);
} while(choice == 'y' || choice == 'Y');
fclose(fp);
printf("\nData successfully stored in file.");
}

OUTPUT

Enter file name: student.txt
Enter data: Hello
Do you want to enter more data? (y/n): y
Enter data: I am a BCA student
Do you want to enter more data? (y/n): y
Enter data: Learning C programming
Do you want to enter more data? (y/n): n
Data successfully stored in file.

#include <stdio.h>
#include <conio.h>
void main()
{
FILE *fp1, *fp2;
char fname1[50], fname2[50], data[100];
clrscr();
printf("Enter file name: ");
scanf("%s", fname1);
fp1 = fopen(fname1, "w");
printf("Enter data: ");
scanf("%s", data);
fprintf(fp1, "%s", data);
fclose(fp1);
printf("Enter new file name: ");
scanf("%s", fname2);
fp1 = fopen(fname1, "r");
fp2 = fopen(fname2, "w");
fscanf(fp1, "%s", data);
fprintf(fp2, "%s", data);
fclose(fp1);
fclose(fp2);
remove(fname1);
printf("File name changed successfully.");
}

Enter data: hey
