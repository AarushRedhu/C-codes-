{
int temp;

scanf("%d", ptr2);
printf("\nBefore Swapping: num1 = %d, num2 = %d\n", *ptr1, *ptr2);
temp = *ptr1;
*ptr1 = *ptr2; *ptr2 = temp;
printf("After Swapping: num1 = %d, num2 = %d\n", *ptr1, *ptr2);
}
void main()
{
int num1, num2;
clrscr();
swap(&num1, &num2);
delay(3000); }

Swap numbers using Call by function

CODE

#include <stdio.h>
#include <conio.h>
#include <dos.h>
void swap(int num1, int num2)
{
int temp;
temp = num1;
num1 = num2;
num2 = temp;
printf("After swapping: num1 = %d, num2 = %d\n", num1, num2);
}
void main()
{
int num1, num2;
clrscr();
printf("Enter num1: ");
scanf("%d", &num1);
printf("Enter num2: ");
scanf("%d", &num2);
printf("\nBefore swapping: num1 = %d, num2 = %d\n", num1, num2);
swap(num1, num2);
delay(3000);
}

OUTPUT

Enter num1: 2
Enter num2: 3
Before swapping: num1 = 2, num2 = 3
After swapping: num1 = 3, num2 = 2
